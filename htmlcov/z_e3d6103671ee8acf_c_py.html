143 def starter(self): 

144 # Retrieve archive links and select valid range 

145 archivePage = self.getPage(self.url) 

146 archiveStrips = self.match(archivePage, '//div[@id="archivepage"]//a') 

147 valid = False 

148 for link in archiveStrips: 

149 if self.stripUrl % '2012/01' in link.get('href'): 

150 valid = True 

151 elif self.stripUrl % '2003/06' in link.get('href'): 

152 valid = False 

153 if valid: 

154 self.strips.append(link.get('href')) 

155 return self.strips.pop(0) 

156 

157 def getPrevUrl(self, url, data): 

158 return self.strips.pop(0)
